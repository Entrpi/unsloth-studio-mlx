# Apple-Silicon MLX backend. Marker limits the pin to macOS so pip
# install on Linux / Windows (where mlx can't build) silently skips it.
mlx-lm>=0.31.2,<0.32 ; sys_platform == "darwin"
# Chunk D (Phase 9): MLX vision-language backend via mlx-vlm.
# Verified against mlx-vlm 0.4.4 (Qwen3.5-VL). Pulls in torch +
# torchvision transitively for the HF AutoProcessor video shim even
# though Studio does not use the video path.
mlx-vlm>=0.4.4,<0.5 ; sys_platform == "darwin"
# Chunk D (Phase 10): MLX audio backend via mlx-audio (LFM2.5-Audio TTS +
# best-effort ASR). mlx-audio 0.4.2 hard-pins mlx-lm==0.31.1 in its
# install_requires but works fine with the >=0.31.2 we pin above — pip
# emits a dependency-resolver warning that is not load-bearing.
mlx-audio>=0.4.2,<0.5 ; sys_platform == "darwin"
# Chunk E (E9): mlx-whisper as a dedicated verbatim-ASR path. LFM2.5-Audio
# transcribes conversationally rather than verbatim, so Whisper is opt-in via
# `backend=whisper` on /v1/audio/transcriptions. Lazy-imported in
# MlxAudioBackend.transcribe_with_whisper so its absence is non-fatal.
mlx-whisper>=0.4.3,<0.5 ; sys_platform == "darwin"
# Chunk H2: cross-platform GPU telemetry. nvidia-ml-py provides the `pynvml`
# module used by core/telemetry/gpu_sampler.py as the primary GPU source on
# Linux / WSL / Windows boxes with an NVIDIA driver (NVIDIA is by far the
# most common inference GPU across Studio's deployment matrix). Marker
# excludes macOS where Apple-Silicon goes through IOReport instead — on
# Intel Macs there's no GPU path anyway. Lazy-imported in the sampler so
# its absence degrades cleanly to the next backend (AMD sysfs, nvidia-smi
# CLI, ...).
nvidia-ml-py>=12.0 ; sys_platform != "darwin"
# Chunk H2: pywin32 provides the `win32pdh` Performance Data Helper used by
# the Windows branch of the GPU sampler (covers AMD / Intel Arc / any-GPU
# on Windows, alongside pynvml for NVIDIA cards). Pulled in only on
# Windows — on Linux/macOS this would be a no-op install.
pywin32>=306 ; sys_platform == "win32"
